# Chinese LLM Provider Dependencies
# ===================================
#
# Usage:
#   pip install -r requirements/cn_models.txt
#
# Or install a specific provider:
#   pip install httpx pyjwt  # Zhipu, Baichuan
#   pip install openai       # DeepSeek, Qwen, Moonshot, Yi

# ===================================
# Core dependencies
# ===================================

# HTTP client - for native APIs such as Zhipu and Baichuan
httpx>=0.24.0

# JWT authentication - for Zhipu API authentication
PyJWT>=2.8.0

# OpenAI SDK - compatible with DeepSeek, Qwen, Moonshot, and Yi
openai>=1.0.0

# ===================================

#
# Configure environment variables:
#   export ZHIPU_API_KEY="your-api-key"
#
# Usage example:
#   from forgedan.adapters import ModelAdapterFactory
#   adapter = ModelAdapterFactory.create_from_string("zhipu:glm-4", api_key="xxx.xxx")
#
# Note: the Zhipu API key format is "api_key_id.api_key_secret"

# ===================================
# Qwen (Qwen / DashScope)
# ===================================
# API docs: https://help.aliyun.com/document_detail/2712195.html
#
# Configure environment variables:
#   export DASHSCOPE_API_KEY="your-api-key"
#
# Usage example:
#   adapter = ModelAdapterFactory.create_from_string("qwen:qwen-max", api_key="xxx")
#
# Supported models:
#   - qwen-turbo: speed first
#   - qwen-plus: balanced performance
#   - qwen-max: best performance
#   - qwen-max-longcontext: 1M context
#   - qwen-vl-max: multimodal

# ===================================

# ===================================

#
# Configure environment variables:
#   export DEEPSEEK_API_KEY="your-api-key"
#
# Usage example:
#   adapter = ModelAdapterFactory.create_from_string("deepseek:deepseek-chat", api_key="xxx")
#
